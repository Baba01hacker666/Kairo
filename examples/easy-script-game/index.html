⚡ EasyScript Game Code

This game is built entirely using Kairo Engine's high-level EasyScript API. Notice how motion, jumping, particle bursts, audio, and AI chasing require 0 boilerplate math! 

// 1. 🏃 Player Controller Script
const playerScript = EasyScript.createBehavior({
  onUpdate(dt) {
    const speed = 6.0;
    if (app.keys.KeyW || app.keys.ArrowUp) this.moveForward(speed * dt);
    if (app.keys.KeyS || app.keys.ArrowBackward) this.moveBackward(speed * dt);
    if (app.keys.KeyA || app.keys.ArrowLeft) this.turnLeft(120 * dt);
    if (app.keys.KeyD || app.keys.ArrowRight) this.turnRight(120 * dt);
    if (app.keys.Space) this.jump(8.0);
  }
});

// 2. 🎈 Collectible Star Item Script
const starScript = EasyScript.createBehavior({
  onStart() {
    this.spin(2.5); // Spins continuously!
    this.bob(0.35, 3.5); // Bobs up and down!
    this.pulse(0.85, 1.2, 3.0); // Pulses size!
  },
  onInteract() {
    this.sparkle(40); // Spawns golden sparkles!
    this.playSound('coin');
    this.say('Collected Golden Star! +100 PTS ⭐', 2000, 'success');
    this.destroy();
  }
});

// 3. 🎯 AI Patrol / Chaser Bot Script
const chaserBotScript = EasyScript.createBehavior({
  onUpdate(dt) {
    if (this.isNear(player.position, 6.0)) {
      this.chase(player.position, 3.2, dt); // Chases player!
    } else {
      this.patrol(5.0, 2.0); // Patrols back and forth!
    }
  },
  onCollision(other) {
    this.explode(50); // Spawns explosion particles!
    this.playSound('explosion');
    this.say('BOOM! Hit by Enemy Patrol Bot! 💥', 2000, 'warning');
  }
});
